    // 获取最近的抢购时间
    function getNextTargetTime() {
        const now = new Date();
        let minDiff = Infinity;
        let nextTarget = null;

        for (const time of targetTimes) {
            const target = new Date();
            target.setHours(time.hour, time.minute, time.second, 0);

            if (target < now) {
                target.setDate(target.getDate() + 1);
            }

            const diff = target - now;
            if (diff < minDiff) {
                minDiff = diff;
                nextTarget = target;
            }
        }

        return { time: nextTarget, diff: minDiff };
    }

    // 格式化时间
    function formatTime(diff) {
        const totalSeconds = Math.floor(diff / 1000);

        if (totalSeconds <= 60) {
            return `${totalSeconds}秒`;
        }

        const hours = Math.floor(totalSeconds / 3600);
        const minutes = Math.floor((totalSeconds % 3600) / 60);
        const seconds = totalSeconds % 60;

        if (hours > 0) {
            return `${hours}小时${minutes}分钟${seconds}秒`;
        } else {
            return `${minutes}分钟${seconds}秒`;
        }
    }

    // 验证卡密并获取核心代码
    async function verifyCard(cardKey) {
        return new Promise((resolve) => {
            // 构建请求URL
            const apiUrl = getVerifyApiUrl();

            // 使用油猴的GM_xmlhttpRequest来绕过混合内容限制
            GM_xmlhttpRequest({
                method: 'POST',
                url: apiUrl,
                headers: {
                    'Content-Type': 'application/x-www-form-urlencoded'
                },
                data: `card=${encodeURIComponent(cardKey)}`,
                timeout: 10000,
                onload: function(response) {
                    if (response.status === 200) {
                        try {
                            const result = JSON.parse(response.responseText);

                            if (result.status === 'success') {
                                // 确保remaining值存在且为数字
                                const remainingCount = parseInt(result.remaining) || 0;
                                resolve({ success: true, core_code: result.core_code, remaining: remainingCount });
                            } else {
                                resolve({ success: false, message: result.message || '卡密验证失败' });
                            }
                        } catch (jsonError) {
                            resolve({ success: false, message: '服务器返回数据格式错误' });
                        }
                    } else {
                        resolve({ success: false, message: '服务器连接失败：HTTP ' + response.status });
                    }
                },
                onerror: function() {
                    resolve({ success: false, message: '无法连接到服务器' });
                },
                ontimeout: function() {
                    resolve({ success: false, message: '服务器连接超时' });
                }
            });
        });
    }

    // 执行核心抢购代码
    function executeCoreCode(coreCode) {
        try {

            showNotification('核心代码执行失败：' + error.message, true);
        }
    }

        updateStatus('下次抢购', formattedTime);

        return diff;
    }

    // 保存卡密
    function saveCardKey(cardKey) {
        GM_setValue('saved_card_key', cardKey);
        GM_setValue('card_saved_at', Date.now());
    }

    // 获取保存的卡密
    function getSavedCardKey() {
        return GM_getValue('saved_card_key', null);
    }

    // 清除保存的卡密
    function clearSavedCardKey() {
        GM_deleteValue('saved_card_key');
        GM_deleteValue('card_saved_at');
    }

    // 添加退出按钮到状态栏
    function addExitButton() {
        const statusWindow = document.getElementById('sniper-status-window');
        if (!statusWindow) return;

        // 检查是否已存在退出按钮
        if (document.getElementById('sniper-exit-button')) return;

        // 创建退出按钮
        const exitButton = document.createElement('button');
        exitButton.id = 'sniper-exit-button';
        exitButton.textContent = '退出';
        exitButton.style.cssText = `
            padding: 4px 8px;
            background-color: #ff4d4f;
            color: white;
            border: none;
            border-radius: 4px;
            font-size: 10px;
            cursor: pointer;
            white-space: nowrap;
        `;

        // 退出按钮点击事件
        async function handleExit() {
            // 清除保存的卡密
            clearSavedCardKey();

            // 清除定时任务和定时器
            if (sniperTimeout) {
                clearTimeout(sniperTimeout);
                sniperTimeout = null;
            }
            if (countdownInterval) {
                clearInterval(countdownInterval);
                countdownInterval = null;
            }

            // 更新状态
            updateStatus('脚本状态', '已停止', true);
            updateStatus('卡密验证', '已退出', true);
            updateStatus('抢购状态', '已取消', true);
            showNotification('已退出卡密并取消抢购任务', true);

            // 移除事件监听器
            exitButton.removeEventListener('click', handleExit);

            // 重新启动脚本
            setTimeout(async () => {
                // 清除状态栏
                const statusWindow = document.getElementById('sniper-status-window');
                if (statusWindow) {
                    // 清理事件监听器
                    if (statusWindow._cleanup) {
                        statusWindow._cleanup();
                    }
                    statusWindow.remove();
                }

                // 重新执行主函数
                await main();
            }, 1000);
        }

        exitButton.addEventListener('click', handleExit);

        // 添加到标题旁边的容器
        const exitButtonContainer = document.getElementById('sniper-exit-button-container');
        if (exitButtonContainer) {
            exitButtonContainer.appendChild(exitButton);
        }
    }

    // 主函数
    async function main() {
        // 检查是否有保存的卡密
        const savedCardKey = getSavedCardKey();
        let cardKey;

        if (savedCardKey) {
            // 自动使用保存的卡密
            cardKey = savedCardKey;
        } else {
            // 获取新卡密
            cardKey = await getCardKey();
        }

        if (!cardKey) {
            return;
        }

        // 验证卡密
        let verificationResult = await verifyCard(cardKey);
        let coreCode;
        let remaining = 0;

        while (!verificationResult.success) {
            // 验证失败，清除保存的卡密
            clearSavedCardKey();

            // 清除定时任务和定时器
            if (sniperTimeout) {
                clearTimeout(sniperTimeout);
                sniperTimeout = null;
            }
            if (countdownInterval) {
                clearInterval(countdownInterval);
                countdownInterval = null;
            }

            // 重新获取卡密，显示错误提示
            cardKey = await getCardKey('卡密验证失败：' + verificationResult.message);
            if (!cardKey) return;

            // 重新验证
            verificationResult = await verifyCard(cardKey);
        }

        // 验证成功
        coreCode = verificationResult.core_code;
        remaining = verificationResult.remaining;

        // 保存卡密
        saveCardKey(cardKey);

        // 检查是否已经创建了状态栏
        if (!document.getElementById('sniper-status-window')) {
            // 创建状态栏
            statusBar = createStatusBar();

            // 初始化状态
            updateStatus('脚本状态', '运行中');
            updateStatus('服务器连接', '成功');
            updateStatus('卡密验证', '成功');
            updateStatus('抢购状态', '等待中');
            updateStatus('下次抢购', '计算中');

            // 显示启动提示
            showNotification('卡密验证成功，准备抢购');

            // 只要卡密剩余次数大于等于1次就添加退出按钮
            if (remaining >= 1) {
                addExitButton();
            }

            // 计算下次抢购时间
            const msToTarget = updateNextSniperTime();

            if (msToTarget > 0) {
                showNotification(`等待抢购时间，还有 ${formatTime(msToTarget)}`);

                // 每秒更新倒计时
                countdownInterval = setInterval(updateNextSniperTime, 1000);

                sniperTimeout = setTimeout(() => {
                    clearInterval(countdownInterval);
                    showNotification('抢购时间已到，立即执行');
                    updateStatus('抢购状态', '执行中');
                    executeCoreCode(coreCode);
                    updateStatus('抢购状态', '已执行');
                }, msToTarget);
            } else {
                showNotification('抢购时间已到，立即执行');
                updateStatus('抢购状态', '执行中');
                executeCoreCode(coreCode);
                updateStatus('抢购状态', '已执行');
            }
        }
    }

    // 启动脚本
    main();
})();
